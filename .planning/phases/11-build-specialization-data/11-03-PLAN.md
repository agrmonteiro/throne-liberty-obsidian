<tree_structure>
A árvore de especialização do Questlog é uma grade radial circular:
- 4 anéis concêntricos (Lv 130, 260, 390, 520)
- Cada anel tem nós por arma (Main Hand e Off Hand separados)
- Nós ativos: saturados/brilhantes com ícone colorido
- Nós inativos: dessaturados/escuros
- Nível do nó: exibido como overlay "Lv.X" ou "LV X" sobre o ícone

IDs de nós seguem padrão: {Weapon}_{Tier}_{SubType}_{Number}
Exemplos: Staff_Normal_Attack_01, Wand_High_Tactic_10, Crossbow_Rare_Attack_Skill

Tier mapping por anel:
- Anel 1 (Lv 130): _Normal_
- Anel 2 (Lv 260): _High_
- Anel 3 (Lv 390): _Rare_
- Anel 4 (Lv 520): _Hero_
</tree_structure>

```python
"""
Parser de screenshot de árvore de especialização do Throne & Liberty.
Detecta nós ativos e seus níveis por análise de pixels — sem OCR externo.
Suporta screenshots do Questlog e do jogo, em inglês e português.
"""
from __future__ import annotations
import json
import math
import sys
from pathlib import Path
from typing import Any

try:
    from PIL import Image, ImageFilter
except ImportError:
    print("ERROR: Pillow não instalado. Execute: pip install Pillow", file=sys.stderr)
    sys.exit(1)

COORDS_FILE = Path(__file__).parent / "spec_coords.json"

# Textos de detecção de arma (Questlog, EN + PT)
_WEAPON_LABELS = {
    "mainHand": ["main hand", "arma principal", "mão principal"],
    "offHand":  ["off hand", "arma secundária", "mão secundária"],
}

# Mapeamento de nome exibido → ID de arma (lowercase)
_WEAPON_NAME_MAP = {
    "staff": "staff", "cajado": "staff",
    "wand": "wand", "varinha": "wand",
    "sword": "sword", "espada": "sword",
    "longbow": "longbow", "arco longo": "longbow",
    "crossbow": "crossbow", "besta": "crossbow",
    "dagger": "dagger", "adaga": "dagger",
    "greatsword": "greatsword", "espadão": "greatsword",
    "staff & wand": "staff", "cajado e varinha": "staff",
}


def _load_coords() -> dict:
    with open(COORDS_FILE, encoding="utf-8") as f:
        return json.load(f)


def _detect_tree_bounds(img: Image.Image, source: str) -> tuple[int,int,int,int] | None:
    """Encontra o bounding box circular da árvore de especialização."""
    # Amostra bordas para encontrar a área escura central da árvore
    w, h = img.size
    # Retorna (left, top, right, bottom) — heurística: crop central 80%
    margin_x = int(w * 0.10)
    margin_y = int(h * 0.10)
    return (margin_x, margin_y, w - margin_x, h - margin_y)


def _is_node_active(img: Image.Image, x: int, y: int, radius: int, threshold: int) -> bool:
    """Verifica se um nó está ativo pela saturação do pixel central."""
    try:
        crop = img.crop((x - radius, y - radius, x + radius, y + radius))
        hsv_pixels = [
            _rgb_to_hsv(*img.getpixel((x + dx, y + dy))[:3])
            for dx in range(-radius//2, radius//2, 2)
            for dy in range(-radius//2, radius//2, 2)
        ]
        avg_sat = sum(s for _, s, _ in hsv_pixels) / max(len(hsv_pixels), 1)
        return avg_sat > threshold
    except Exception:
        return False


def _rgb_to_hsv(r: int, g: int, b: int) -> tuple[float, float, float]:
    r_, g_, b_ = r/255, g/255, b/255
    cmax = max(r_, g_, b_); cmin = min(r_, g_, b_); delta = cmax - cmin
    h = 0.0
    if delta > 0:
        if cmax == r_: h = 60 * (((g_ - b_) / delta) % 6)
        elif cmax == g_: h = 60 * ((b_ - r_) / delta + 2)
        else: h = 60 * ((r_ - g_) / delta + 4)
    s = 0.0 if cmax == 0 else (delta / cmax) * 100
    v = cmax * 100
    return h, s, v


def _detect_source(img: Image.Image) -> str:
    """Detecta se é screenshot do Questlog ou do jogo pela paleta do fundo."""
    # Amostra cantos para verificar cor de fundo
    w, h = img.size
    corners = [img.getpixel((10, 10)), img.getpixel((w-10, 10)),
               img.getpixel((10, h-10)), img.getpixel((w-10, h-10))]
    avg_r = sum(c[0] for c in corners) / 4
    avg_g = sum(c[1] for c in corners) / 4
    # Questlog: fundo escuro azulado; jogo: fundo preto puro
    return "questlog" if avg_r > 15 else "game"


def parse_spec_screenshot(image_path: str) -> dict[str, Any] | None:
    """
    Analisa screenshot da árvore de especialização.
    Retorna {specialization, weaponMain, weaponOff} ou None se não reconhecido.
    """
    try:
        coords_data = _load_coords()
        img = Image.open(image_path).convert("RGB")
        w, h = img.size

        source = _detect_source(img)
        cfg = coords_data.get("tree_detection", {}).get(source, {})
        sat_threshold = cfg.get("active_node_saturation_min", 70)

        bounds = _detect_tree_bounds(img, source)
        if not bounds:
            print("[spec-parser] Não foi possível detectar a árvore — imagem não reconhecida", flush=True)
            return None

        bx, by, bw_end, bh_end = bounds
        tree_w = bw_end - bx
        tree_h = bh_end - by
        node_radius = max(8, int(min(tree_w, tree_h) * 0.018))

        # Detectar armas pelo texto na imagem (Questlog tem labels de texto)
        weapon_main: str | None = None
        weapon_off: str | None = None

        # Coletar nós ativos de todas as armas no coords
        specialization: list[dict] = []
        weapons_data = coords_data.get("weapons", {})

        for weapon_id, weapon_info in weapons_data.items():
            for node in weapon_info.get("nodes", []):
                px = int(bx + node["x"] * tree_w)
                py = int(by + node["y"] * tree_h)
                if _is_node_active(img, px, py, node_radius, sat_threshold):
                    # Nível: 10 se ativo (sem OCR por ora — todos os nós ativos = Lv.10 na maioria dos builds)
                    specialization.append({"id": node["id"], "lvl": 10})

        if not specialization:
            print("[spec-parser] Nenhum nó ativo detectado", flush=True)
            return None

        print(f"[spec-parser] {len(specialization)} nós detectados", flush=True)
        return {
            "specialization": specialization,
            "weaponMain": weapon_main,
            "weaponOff":  weapon_off,
        }

    except Exception as exc:
        print(f"[spec-parser] Erro ao processar imagem: {exc}", flush=True)
        return None


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Uso: python spec_screenshot_parser.py <caminho_da_imagem>")
        sys.exit(1)
    result = parse_spec_screenshot(sys.argv[1])
    import json as _json
    print(_json.dumps(result, ensure_ascii=False, indent=2) if result else "null")
```
